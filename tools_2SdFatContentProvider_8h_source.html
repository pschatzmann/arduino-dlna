75 void serveFile(IClientHandler& client, const char* requestPath) {
76 StrView path_str{requestPath};
77 int id_pos = path_str.indexOf("ID=");
78 long id = id_pos >= 0 ? atol(requestPath + id_pos + 3) : -1;
79 // id 0 is the root directory
80 if (p_fs == nullptr || id <= 0 || id >= (long)size()) {
81 DlnaLogger.log(DlnaLogLevel::Error, "Invalid file request: %s",
82 requestPath);
83 client.replyNotFound();
84 return;
85 }
86 TreeNode& node = getTreeNodeById(id);
87 if (node.is_dir) {
88 client.replyNotFound();
89 return;
90 }
91 // keep the std::string alive: getMime() returns a temporary
92 std::string mime = node.getMime();
93 if (mime.empty()) mime = "application/octet-stream";
94 std::string file_name = node.path();
95 DlnaLogger.log(DlnaLogLevel::Info, "file: %s (mime %s)",
96 file_name.c_str(), mime.c_str());
97
98 auto file = p_fs->open(file_name.c_str());
99 if (!file) {
100 DlnaLogger.log(DlnaLogLevel::Error, "Failed to open file: %s (id %ld)",
101 file_name.c_str(), id);
102 client.replyNotFound();
103 return;
104 }
105 replyFile(client, file, mime.c_str());
106 file.close();
107 }

114 template <typename F>
115 static void replyFile(IClientHandler& client, F& file, const char* mime) {
116 uint64_t size = file.size();
117 HttpRange range;
118 if (range.parse(client.requestHeader().get(RANGE_C), size)) {
119 if (!range.isSatisfiable()) {
120 client.replyRangeNotSatisfiable(size);
121 return;
122 }
123 file.seek(range.start());
124 client.replyRange(mime, file, size, range.start(), range.end());
125 return;
126 }
127 // full content: HEAD requests are also handled here (header only)
128 client.replyHeader().put(ACCEPT_RANGES, "bytes");
129 client.reply(mime, file, size, 200);
130 }

158 void prepareData(const char* objectID, ContentQueryType queryType,
159 const char* filter, int startingIndex, int requestedCount,
160 const char* sortCriteria, int& numberReturned,
161 int& totalMatches, int& updateID, void* reference) {
162 // Store parameters in class variables
163 objectID_ = objectID;
164 queryType_ = queryType;
165 filter_ = filter;
166 startingIndex_ = startingIndex;
167 requestedCount_ = requestedCount;
168 sortCriteria_ = sortCriteria;
169 reference_ = reference;
170
171 // Clear previous results
172 resultNodes_.clear();
173
174 // Calculate actual values based on directory tree and query type
175 updateID = 0; // Can increment this when content changes
176
177 if (queryType == ContentQueryType::BrowseMetadata) {
178 // Metadata for a single object
179 TreeNode* node =
180 objectID ? findNodeByObjectID(objectID) : &directoryTree.root();
181 if (node && !isHidden(node)) {
182 resultNodes_.push_back(node);
183 totalMatches = 1;
184 numberReturned = 1;
185 } else {
186 totalMatches = 0;
187 numberReturned = 0;
188 }
189 } else if (queryType == ContentQueryType::BrowseChildren) {
190 // Browse children of a container (directory)
191 TreeNode* parentNode =
192 objectID ? findNodeByObjectID(objectID) : &directoryTree.root();
193 if (parentNode) {
194 // Collect all children, filter out hidden
195 resultNodes_.clear();
196 for (auto* child : parentNode->children) {
197 if (!isHidden(child)) {
198 resultNodes_.push_back(child);
199 }
200 }
201 totalMatches = static_cast<int>(resultNodes_.size());
202
203 // Apply pagination
204 int available = totalMatches - startingIndex;
205 if (available < 0) available = 0;
206 numberReturned = (requestedCount == 0)
207 ? available
208 : std::min(requestedCount, available);
209 } else {
210 totalMatches = 0;
211 numberReturned = 0;
212 }
213 } else {
214 // Search - return all files, filter out hidden
215 auto allFiles = directoryTree.getAllFiles();
216 resultNodes_.clear();
217 for (auto* node : allFiles) {
218 if (!isHidden(node)) {
219 resultNodes_.push_back(node);
220 }
221 }
222 totalMatches = static_cast<int>(resultNodes_.size());
223
224 // Apply pagination
225 int available = totalMatches - startingIndex;
226 if (available < 0) available = 0;
227 numberReturned = (requestedCount == 0)
228 ? available
229 : std::min(requestedCount, available);
230 }
231
232 DlnaLogger.log(DlnaLogLevel::Info,
233 "Number of results: Total=%d, Returned=%d", totalMatches,
234 numberReturned);
235 }

257 bool getData(int index, MediaItem& item, void* reference = nullptr) {
258 // Calculate the actual index into resultNodes_
259 int adjustedIndex = startingIndex_ + index;
260
261 // Validate index
262 if (adjustedIndex < 0 ||
263 adjustedIndex >= static_cast<int>(resultNodes_.size())) {
264 return false;
265 }
266
267 TreeNode* node = resultNodes_[adjustedIndex];
268 if (!node) return false;
269
270 // Populate MediaItem from TreeNode
271 // Use member strings for pointer lifetime
272 currentId_ = std::to_string(node->id);
273 item.id = currentId_.c_str();
274
275 // Parent ID
276 if (node->parent) {
277 currentParentId_ = std::to_string(node->parent->id);
278 item.parentID = currentParentId_.c_str();
279 } else {
280 item.parentID = "0";
281 }
282
283 item.title = node->file_name.c_str();
284 item.restricted = true;
285
286 if (node->is_dir) {
287 item.itemClass = MediaItemClass::Folder;
288 item.resourceURI = nullptr;
289 item.mimeType = nullptr;
290 } else {
291 // File - determine class from MIME type
292 currentMime_ = node->getMime();
293 if (currentMime_.empty()) currentMime_ = "application/octet-stream";
294 item.mimeType = currentMime_.c_str();
295
296 if (currentMime_.find("audio") != std::string::npos) {
297 item.itemClass = MediaItemClass::Music;
298 } else if (currentMime_.find("video") != std::string::npos) {
299 item.itemClass = MediaItemClass::Video;
300 } else if (currentMime_.find("image") != std::string::npos) {
301 item.itemClass = MediaItemClass::Photo;
302 } else {
303 item.itemClass = MediaItemClass::Unknown;
304 }
305
306 // Resource URI: baseURL + basePath + ?ID=id
307 char urlBuffer[80];
308 snprintf(urlBuffer, sizeof(urlBuffer), "/%s?ID=%u", path, node->id);
309 currentPath_ = urlBuffer;
310 item.resourceURI = currentPath_.c_str();
311 }
312
313 item.albumArtURI = nullptr;
314
315 return true;
316 }
